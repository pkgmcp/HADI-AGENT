Practical code examples covering real-world use cases.

1. Basic MCP Chat

Send a message and get an AI response via the MCP server:

phpuse App\DTOs\MCPMessageDTO;
use App\Services\MCP\MCPServer;

$server = app(MCPServer::class);

$response = $server->handle(
    new MCPMessageDTO(
        role: 'user',
        content: 'Explain Laravel service providers in simple terms',
        metadata: [
            'provider' => 'openai',
            'temperature' => 0.5,
        ],
    )
);

echo "Response: {$response->content}\n";
echo "Tokens: {$response->totalTokens()} ({$response->inputTokens} in / {$response->outputTokens} out)\n";
echo "Cost: \${$response->cost}\n";
echo "Latency: {$response->latencyMs}ms\n";

2. Multi-Provider Fallback

Configure automatic fallback when the primary provider fails:

php// config/mcp.php
'default_provider' => 'openai',

// .env
AI_FALLBACK_ENABLED=true
AI_FALLBACK_PROVIDER=ollama    # Falls back to local Ollama

// Usage — automatic fallback
$response = $server->handle(
    new MCPMessageDTO(
        role: 'user',
        content: 'Write a PHP function to validate email',
        metadata: ['provider' => 'nonexistent'],  // Will fail → fallback
    )
);

// Or use ProviderRouter directly
$router = app(\App\Services\MCP\ProviderRouter::class);
$response = $router->routeWithFallback($request);

3. Agent-Based Code Generation Pipeline

Use multiple agents in sequence to generate and review code:

phpuse App\Services\Agents\{PlannerAgent, CoderAgent, ReviewAgent, SecurityAgent};

// Step 1: Plan
$planner = app(PlannerAgent::class);
$plan = $planner->createExecutionPlan(
    'Build a Laravel REST API with authentication',
    ['framework' => 'Laravel 12', 'features' => ['auth', 'crud', 'api']]
);

// Step 2: Generate code
$coder = app(CoderAgent::class);
foreach ($plan['files'] as $file) {
    $result = $coder->generateFile($file['path'], $file['spec']);
    file_put_contents(base_path($file['path']), $result['content']);
}

// Step 3: Review
$reviewer = app(ReviewAgent::class);
$review = $reviewer->reviewCode(
    file_get_contents(base_path('app/Http/Controllers/Api/AuthController.php')),
    'php'
);

// Step 4: Security audit
$security = app(SecurityAgent::class);
$audit = $security->auditCode(
    file_get_contents(base_path('app/Http/Controllers/Api/AuthController.php'))
);

echo "Review findings: " . count($review['findings']) . "\n";
echo "Security issues: " . count($audit['vulnerabilities']) . "\n";

4. Workflow-Driven Code Review

A complete workflow that analyzes, reviews, and reports on code quality:

phpuse App\DTOs\WorkflowDTO;
use App\Services\MCP\WorkflowEngine;

$engine = app(WorkflowEngine::class);

$code = file_get_contents(base_path('app/Models/User.php'));

$dto = new WorkflowDTO('Complete Code Review', [
    [
        'name' => 'analyze',
        'type' => 'prompt',
        'prompt' => 'Analyze this Laravel code and provide a summary: {{context.code}}',
    ],
    [
        'name' => 'security_review',
        'type' => 'agent',
        'agent' => 'security',
        'task' => 'Audit this code: {{results.analyze.content}}',
    ],
    [
        'name' => 'quality_check',
        'type' => 'prompt',
        'prompt' => 'Rate code quality 1-10 and suggest improvements: {{context.code}}',
    ],
    [
        'name' => 'generate_report',
        'type' => 'prompt',
        'prompt' => 'Create a markdown report combining these findings:
            Analysis: {{results.analyze.content}}
            Security: {{results.security_review.content}}
            Quality: {{results.quality_check.content}}',
    ],
], context: ['code' => $code]);

$id = $engine->create($dto);
$result = $engine->execute($id);

// Save report
file_put_contents(
    storage_path('reports/code-review-' . now()->timestamp . '.md'),
    $result['results']['generate_report']['content']
);

echo "Review complete: {$result['status']}\n";

5. Token Usage Dashboard (PHP)

Generate a cost summary using the TokenObserver and CostAnalyzer:

phpuse App\Services\MCP\{TokenObserver, CostAnalyzer};

$observer = app(TokenObserver::class);
$analyzer = app(CostAnalyzer::class);

// Get daily summary
$dailySummary = $observer->summary();
echo "Daily tokens: {$dailySummary['daily_tokens']}\n";
echo "Active sessions: {$dailySummary['total_sessions']}\n";

// Cost breakdown
$costSummary = $analyzer->summary();
echo "Total cost today: \${$costSummary['total']}\n";
echo "By provider:\n";
foreach ($costSummary['by_provider'] as $provider => $cost) {
    echo "  - {$provider}: \${$cost}\n";
}

// Budget check
$budget = $analyzer->budgetUtilization();
echo "Budget: \${$budget['budget']}\n";
echo "Spent: \${$budget['spent']}\n";
echo "Remaining: \${$budget['remaining']}\n";
echo "Utilization: {$budget['utilization_percent']}%\n";

// 30-day projection
$projection = $analyzer->projection(30);
echo "30-day projection: \${$projection}\n";

6. Custom AI Provider

Integrate your own LLM by implementing the provider interface:

phpuse App\Interfaces\AIProviderInterface;
use App\DTOs\{AIRequestDTO, AIResponseDTO};

class CustomProvider implements AIProviderInterface
{
    public function send(AIRequestDTO $request): AIResponseDTO
    {
        // Call your custom LLM API
        $response = Http::post('https://your-llm-api.com/generate', [
            'prompt' => $request->prompt,
            'max_tokens' => $request->maxTokens,
        ]);

        return new AIResponseDTO(
            content: $response->json('text'),
            provider: 'custom-llm',
            model: $request->model,
            inputTokens: $request->maxTokens,
            outputTokens: strlen($response->json('text')),
            cost: 0.001,
            latencyMs: 200,
        );
    }

    public function stream(AIRequestDTO $request): iterable { yield ''; }
    public function name(): string { return 'custom-llm'; }
    public function isAvailable(): bool { return true; }
    public function models(): array { return ['custom-v1']; }
    public function calculateCost(int $in, int $out): float { return 0.001; }
}

// Register it
$server = app(\App\Services\MCP\MCPServer::class);
$server->registerProvider('custom-llm', new CustomProvider);

7. Memory-Enhanced Chat

Use MemoryAgent to maintain context across sessions:

php$memory = app(\App\Services\Agents\MemoryAgent::class);
$server = app(\App\Services\MCP\MCPServer::class);

$sessionId = 'user_chat_' . auth()->id();

// Store conversation history
$memory->store("session_{$sessionId}", [
    'context' => $server->handle(
        new MCPMessageDTO('user', 'My name is John', metadata: ['session_id' => $sessionId])
    )->content,
]);

// Later — retrieve context
$previousContext = $memory->retrieve("session_{$sessionId}");
$memory->store("session_{$sessionId}", [
    'context' => $previousContext['context'] . "\n" .
        $server->handle(new MCPMessageDTO('user', 'What is my name?', metadata: ['session_id' => $sessionId])
    )->content,
]);

// Search memory
$results = $memory->search('John');
echo "Found " . count($results) . " relevant memories\n";

8. Environment Validation (Deployment Agent)

php$deployment = app(\App\Services\Agents\DeploymentAgent::class);

// Validate your Laravel .env
$validation = $deployment->validateEnvironment([
    'APP_ENV' => 'production',
    'APP_DEBUG' => false,
    'DB_CONNECTION' => 'mysql',
    'REDIS_HOST' => '127.0.0.1',
    'QUEUE_CONNECTION' => 'redis',
    'SESSION_DRIVER' => 'redis',
    'CACHE_STORE' => 'redis',
]);

if (!empty($validation['issues'])) {
    foreach ($validation['issues'] as $issue) {
        echo "[{$issue['severity']}] {$issue['message']}\n";
    }
}

// Generate Dockerfile
$dockerfile = $deployment->generateDockerfile(
    'laravel',
    ['php' => '8.3', 'node' => '20', 'database' => 'postgresql']
);
file_put_contents(base_path('Dockerfile'), $dockerfile);

9. cURL API Examples

curl http://localhost:8000/api/v1/health

  -d '{"task": "Plan a blog platform API"}'

# Create workflow
curl -X POST http://localhost:8000/api/v1/workflows \
  -H "Content-Type: application/json" \
  -d '{
    "name": "Test WF",
    "steps": [{"name": "s1", "type": "prompt", "prompt": "Hello"}]
  }'

# Execute workflow
curl -X POST http://localhost:8000/api/v1/workflows/{wf_id}/execute

# Check status
curl http://localhost:8000/api/v1/workflows/{wf_id}/status
